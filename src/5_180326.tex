\begin{theorem}[Charateodory]

    Let $X \seq \Rn$ be nonempty. Then
    
    \begin{enumerate}
        \item[a)] every nonzero vector from $\cone{X}$ can be represented as a positive combination of linearly independent vectors in $X$;
        \item[b)] every vector fron $\conv{X}$ can be represented as a convex combination of no more then $n+1$ linearly independent vectors in $X$.
    \end{enumerate}
    
\end{theorem}
